cmake_minimum_required(VERSION 3.24)
project(ninfer_cpu_vision_tests LANGUAGES C CXX)

# Standalone CPU-only admission/numerical tests: never enables CUDA detection.
get_filename_component(NINFER_SOURCE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
option(NINFER_BUILD_CPU_VISION "Build the pinned GGML CPU vision backend" ON)
include("${NINFER_SOURCE_ROOT}/cmake/ggml_cpu_vision.cmake")
enable_testing()
add_executable(test_cpu_vision_cache "${NINFER_SOURCE_ROOT}/tests/tests_cpu_vision_cache.cpp")
target_link_libraries(test_cpu_vision_cache PRIVATE ninfer_cpu_vision)
add_test(NAME cpu_vision_cache COMMAND test_cpu_vision_cache)
add_executable(test_cpu_vision_bridge "${NINFER_SOURCE_ROOT}/tests/targets/qwen3_6/test_cpu_vision_bridge.cpp")
target_link_libraries(test_cpu_vision_bridge PRIVATE ninfer_cpu_vision)
add_test(NAME cpu_vision_bridge COMMAND test_cpu_vision_bridge)

add_executable(test_cpu_vision_startup "${NINFER_SOURCE_ROOT}/tests/targets/qwen3_6/test_startup_features.cpp")
target_include_directories(test_cpu_vision_startup PRIVATE "${NINFER_SOURCE_ROOT}/include"
    "${NINFER_SOURCE_ROOT}/src/targets/qwen3_6/export")
target_link_libraries(test_cpu_vision_startup PRIVATE ninfer_cpu_vision)
add_test(NAME cpu_vision_startup COMMAND test_cpu_vision_startup)
add_executable(test_cpu_vision_cli "${NINFER_SOURCE_ROOT}/tests/test_cli_options.cpp"
    "${NINFER_SOURCE_ROOT}/apps/cli/options.cpp")
target_include_directories(test_cpu_vision_cli PRIVATE "${NINFER_SOURCE_ROOT}/include")
target_link_libraries(test_cpu_vision_cli PRIVATE ninfer_cpu_vision)
add_test(NAME cpu_vision_cli COMMAND test_cpu_vision_cli)

add_executable(test_ggml_cpu_vision "${NINFER_SOURCE_ROOT}/tests/targets/qwen3_6/test_ggml_cpu_vision.cpp")
target_link_libraries(test_ggml_cpu_vision PRIVATE ninfer_cpu_vision)
if(NINFER_BUILD_CPU_VISION)
    add_executable(bench_cpu_vision bench_cpu_vision.cpp)
    target_link_libraries(bench_cpu_vision PRIVATE ninfer_cpu_vision)
    if(WIN32)
        target_link_libraries(bench_cpu_vision PRIVATE psapi)
    endif()
    target_compile_definitions(test_ggml_cpu_vision PRIVATE NINFER_TEST_GGML_CPU_VISION=1)
    target_link_libraries(test_ggml_cpu_vision PRIVATE mtmd)
    add_test(NAME ggml_cpu_vision_numerics COMMAND test_ggml_cpu_vision --numerics)
    add_test(NAME ggml_cpu_vision_real_gguf COMMAND test_ggml_cpu_vision --gguf)
    set_tests_properties(ggml_cpu_vision_real_gguf PROPERTIES SKIP_RETURN_CODE 77)
else()
    add_test(NAME cpu_vision_not_built COMMAND test_ggml_cpu_vision --stub)
endif()
